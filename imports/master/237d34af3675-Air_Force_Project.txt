# AUTHENTICATION (Badge + PIN — upgrade path to CAC noted)
# ════════════════════════════════════════════════════════════════════════════════
# NOTE FOR GOOGLE AI: This is a minimal credential check using hashed PINs
# stored in st.secrets. Production upgrade = CAC/PIV + LDAP/AD integration.
# The session token ties all audit entries to a verified identity.

DEMO_USERS = {
    "AF-001": hashlib.sha256(b"1234").hexdigest(),
    "AF-002": hashlib.sha256(b"5678").hexdigest(),
    "GUEST":  hashlib.sha256(b"0000").hexdigest(),
}

def authenticate(badge: str, pin: str) -> bool:
    pin_hash = hashlib.sha256(pin.encode()).hexdigest()
    stored   = DEMO_USERS.get(badge.upper())
    if stored and stored == pin_hash:
        st.session_state.authenticated = True
        st.session_state.session_user  = badge.upper()
        log_action(badge.upper(), "SESSION", "LOGIN", "", f"Session {st.session_state.session_id}")
        return True
    return False

# ════════════════════════════════════════════════════════════════════════════════
# AUDIT LOG — Immutable Append-Only (tamper-evident via entry hash chain)
# ════════════════════════════════════════════════════════════════════════════════
def log_action(user: str, doc_id: str, action: str, original: str, final: str):
    prev_hash = (
        st.session_state.audit_log[-1]["entry_hash"]
        if st.session_state.audit_log else "GENESIS"
    )
    entry = {
        "entry_id":     str(uuid.uuid4()),
        "timestamp":    datetime.utcnow().strftime("%Y-%m-%d %H:%M:%S UTC"),
        "session_id":   st.session_state.session_id,
        "user":         user,
        "doc_id":       doc_id,
        "action":       action,
        "original":     original[:500],   # Truncate for storage
        "final":        final[:500],
        "prev_hash":    prev_hash,
    }
    # Chain hash — each entry proves previous wasn't altered
    entry_str  = json.dumps({k: entry[k] for k in entry if k != "entry_hash"}, sort_keys=True)
    entry["entry_hash"] = hashlib.sha256(entry_str.encode()).hexdigest()[:16]
    st.session_state.audit_log.append(entry)

    # Persist to Supabase if available
    if SUPABASE:
        try:
            SUPABASE.table("audit_log").insert(entry).execute()
        except Exception:
            pass   # Fail silently — in-memory log is always the source of truth

# ════════════════════════════════════════════════════════════════════════════════
# FILE EXTRACTION — PDF / DOCX / TXT
# ════════════════════════════════════════════════════════════════════════════════
def extract_text(file) -> str:
    try:
        ftype = file.type
        if "pdf" in ftype and PDF_OK:
            reader = PdfReader(file)
            return "\n".join([p.extract_text() or "" for p in reader.pages])
        if "word" in ftype and DOCX_OK:
            doc = docx.Document(file)
            return "\n".join([p.text for p in doc.paragraphs])
        # Plain text fallback
        raw = file.read()
        return raw.decode("utf-8", errors="replace")
    except Exception as e:
        return f"[EXTRACTION ERROR: {e}]"

# ════════════════════════════════════════════════════════════════════════════════
# COMPLIANCE ANALYSIS ENGINE
# Returns list of findings with context-awareness (checks safe_exceptions)
# ════════════════════════════════════════════════════════════════════════════════
def analyze_text(text: str) -> list:
    findings = []
    text_lower = text.lower()
    for rule in FAR_RULES:
        matches = list(re.finditer(rule["pattern"], text, re.IGNORECASE))
        if not matches:
            continue
        # Filter safe exceptions
        filtered = []
        for m in matches:
            context_snippet = text_lower[max(0, m.start()-30):m.end()+30]
            is_safe = any(exc in context_snippet for exc in rule.get("safe_exceptions", []))
            if not is_safe:
                filtered.append(m)
        if filtered:
            findings.append({
                "rule":       rule,
                "matches":    filtered,
                "count":      len(filtered),
            })
    return findings
